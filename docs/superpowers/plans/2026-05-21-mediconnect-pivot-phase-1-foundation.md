# MediConnect Pivot — Phase 1: Foundation & Teardown

> **⚠️ SUPERSEDED (2026-05-22):** This phase planned a from-scratch refactor of the Amelia frontend that lived at `nalavita-frontend/`. That frontend has been replaced wholesale by a pre-built MediConnect implementation relocated from `C:\Users\IKA\Med connect\` to the Nala Vita root. **Do not execute the tasks below.** They describe a starting state (Amelia codebase) that no longer exists in this repo. See `…roadmap.md` superseded banner for what's still actionable.

> **For agentic workers:** REQUIRED SUB-SKILL: Use **superpowers:subagent-driven-development** (recommended) or **superpowers:executing-plans** to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. Each task is one git commit.

**Goal:** Reshape `nalavita-frontend/` to match the MediConnect spec's skeleton — full schema, route groups, lib helpers, RBAC middleware, test harness — and delete the Amelia AI / Gemini surface. At the end of this phase the app boots, `/login` renders, unauthenticated users are redirected to `/login`, and the new Prisma schema is migrated. No business logic yet — that lands in later phases.

**Architecture:** Single migration commit per concern. Tests are written for everything that has logic (zod schemas, `createNotification`, RBAC middleware). Pure config / structural moves (deps install, file renames, schema changes) skip TDD but have explicit verification commands. We switch from Prisma `db push` to `prisma migrate dev` so future schema changes are tracked.

**Tech Stack:** Next.js 16 (App Router), TypeScript, Tailwind 4, Prisma 7, Supabase (`@supabase/ssr`), Vitest (new), zod, OpenAI SDK.

**Working directory for all commands:** `C:\Users\IKA\Nala Vita\nalavita-frontend\` unless stated otherwise.

// Server-side Supabase client. Used in API routes and Server Components.
// Replaces spec's createServerComponentClient (deprecated) with @supabase/ssr.
import { createServerClient as createSSRServerClient, type CookieOptions } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createServerClient() {
  const cookieStore = await cookies();
  return createSSRServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll();
        },
        setAll(cookiesToSet) {
          try {
            cookiesToSet.forEach(({ name, value, options }: { name: string; value: string; options: CookieOptions }) =>
              cookieStore.set(name, value, options)
            );
          } catch {
            // Server Components can't set cookies — middleware handles refresh.
          }
        },
      },
    }
  );
}

// Convenience wrapper for routes that just need the authed user.
export async function getAuthenticatedUser() {
  const supabase = await createServerClient();
  const { data: { user } } = await supabase.auth.getUser();
  return user;
}

```typescript
import { NextRequest, NextResponse } from "next/server";
import { createServerClient } from "@supabase/ssr";

const PUBLIC_PATHS = [
  "/login",
  "/register",
  "/unauthorized",
  "/forgot-password",
  "/reset-password",
  "/favicon.ico",
];

const PUBLIC_PREFIXES = ["/_next/", "/api/auth/", "/api/health"];

function isPublic(pathname: string): boolean {
  if (PUBLIC_PATHS.includes(pathname)) return true;
  return PUBLIC_PREFIXES.some((p) => pathname.startsWith(p));
}

function roleForPath(pathname: string): "PATIENT" | "DOCTOR" | "ADMIN" | null {
  if (pathname.startsWith("/patient")) return "PATIENT";
  if (pathname.startsWith("/doctor")) return "DOCTOR";
  if (pathname.startsWith("/admin")) return "ADMIN";
  return null;
}

export async function middleware(req: NextRequest) {
  const { pathname } = req.nextUrl;

  if (isPublic(pathname)) {
    return NextResponse.next();
  }

  const res = NextResponse.next();

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return req.cookies.getAll();
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value, options }) => {
            res.cookies.set(name, value, options);
          });
        },
      },
    }
  );

  const { data: { user } } = await supabase.auth.getUser();

  if (!user) {
    const url = req.nextUrl.clone();
    url.pathname = "/login";
    return NextResponse.redirect(url);
  }

  const requiredRole = roleForPath(pathname);
  if (requiredRole) {
    const actualRole = (user.user_metadata?.role ?? "PATIENT") as string;
    if (actualRole !== requiredRole) {
      const url = req.nextUrl.clone();
      url.pathname = "/unauthorized";
      return NextResponse.redirect(url);
    }
  }

  return res;
}

export const config = {
  matcher: [
    /*
     * Match all paths except:
     * - _next/static, _next/image, favicon.ico
     * - public assets
     */
    "/((?!_next/static|_next/image|favicon.ico).*)",
  ],
};
```

import { useEffect, useState } from "react";
import type { User } from "@supabase/supabase-js";

export function useAuth() {
  const [user, setUser] = useState<User | null>(null);

}
